488void *dentist(void *arg)
489{

504 {
505 pthread_mutex_unlock(&mutex);
506 break;
507 }
508
509 /* Doctor has treatment capacity */
510 if (waitingPatients > 0 &&
511 treatedPatients < doctorCapacity)
512 {
513 int id = findNextPatient();
514
515 if (id != -1)
516 {
517 patients[id].waiting = 0;
518 waitingPatients--;

528 waitingPatients,
529 waitingRoomChairs);
530
531 pthread_mutex_unlock(&mutex);
532
533 /* Call exact patient */
534 sem_post(&patients[id].dentistCallSem);

536 printf("[Dr. Smith] Treating %s...\n",
537 patients[id].name);
538
539 /* Treatment simulation */
540 sleep(2);
541
542 printf("[Dr. Smith] Treatment finished for %s.\n",

546

550 /* Release appointment slot */
551 if (patients[id].appointment == 1 &&

554 activeAppointments--;
555 }

557 pthread_mutex_unlock(&mutex);
558
559 /* Notify patient */
560 sem_post(&patients[id].treatmentDoneSem);

581
582 waitingPatients--;
583

590 }
591 }

593 pthread_mutex_unlock(&mutex);

597 pthread_mutex_unlock(&mutex);
598 }
